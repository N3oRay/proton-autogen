OPCIONES
━━━━━━━━━━━━━━━━━━━━━━━━━━
--debug       Activar el modo de depuración
--verbose     Activar el modo detallado
--mangohud    Mostrar los FPS y la información de rendimiento
--gamemode    Optimizar el rendimiento del sistema
--gamescope   Utilizar Gamescope
--wine        Forzar el uso de Wine
--proton      Forzar el uso de Proton

━━━━━━━━━━━━━━━━━━━━━━━━━━
EJEMPLOS
━━━━━━━━━━━━━━━━━━━━━━━━━━
proton-autogen game.exe
proton-autogen game.exe --gamemode --mangohud --gamescope

# Gamescope integrado
proton-autogen game.exe --gamescope

# Gamescope manual (avanzado)
gamescope -f -W 1280 -H 1024 -- proton-autogen game.exe
